### docker-compose-lite.yml 参考结构

```yaml
services:
  mysql:
    image: mysql:8.0
    environment:
      MYSQL_ROOT_PASSWORD: rootmarschat
      MYSQL_DATABASE: knowledge_base
      MYSQL_USER: kb
      MYSQL_PASSWORD: kb123456
    ports: ["3306:3306"]
    volumes:
      - /data/mysql:/var/lib/mysql
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql
    command: --character-set-server=utf8mb4 --collation-server=utf8mb4_unicode_ci
    deploy: { resources: { limits: { memory: 512M } } }

  mongo:
    image: mongo:7.0
    ports: ["27017:27017"]
    volumes:
      - /data/mongodb:/data/db
    deploy: { resources: { limits: { memory: 512M } } }

  redis:
    image: redis:7-alpine
    ports: ["6379:6379"]
    command: redis-server --maxmemory 128mb --maxmemory-policy allkeys-lru
    deploy: { resources: { limits: { memory: 128M } } }

  minio:
    image: minio/minio:latest
    ports: ["9000:9000", "9001:9001"]
    environment:
      MINIO_ROOT_USER: minioadmin
      MINIO_ROOT_PASSWORD: minioadmin
    volumes:
      - /mnt/e-data/minio:/data
    command: server /data --console-address ":9001"
    deploy: { resources: { limits: { memory: 512M } } }

  meilisearch:
    image: getmeili/meilisearch:latest
    ports: ["7700:7700"]
    environment:
      MEILI_MASTER_KEY: masterKey
    volumes:
      - /data/meilisearch:/meili_data
    deploy: { resources: { limits: { memory: 200M } } }

  kb-backend:
    build: .
    ports: ["8080:8080"]
    environment:
      SPRING_PROFILES_ACTIVE: prod
      DB_PASSWORD: kb123456
      REDIS_PASSWORD: ""
    depends_on: [mysql, mongo, redis, minio, meilisearch]
    deploy: { resources: { limits: { memory: 512M } } }
```

### Nginx配置（完整）

```nginx
# /kb/api/ → VM内Spring Boot
location /kb/api/ {
    proxy_pass http://旧Windows_Tailscale_IP:8080/kb/api/;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_connect_timeout 5s;
    proxy_read_timeout 60s;
    client_max_body_size 200m;
}

# /kb/s/ → 静态资源
location /kb/s/assets/ {
    alias /data/knowledge-base/web/assets/;
    expires 1y;
    add_header Cache-Control "public, immutable";
}

location /kb/s/ {
    alias /data/knowledge-base/web/;
    expires 7d;
}

# /kb/share/ → SPA（无需登录）
location /kb/share/ {
    try_files $uri /kb/s/index.html;
}

# /kb/ → SPA fallback
location /kb/ {
    try_files $uri /kb/s/index.html;
}

# 根路径重定向
location = / {
    return 302 /kb/;
}

location = /kb {
    return 302 /kb/;
}
```

### 备份脚本参考

```bash
#!/bin/bash
# mysql-backup.sh
DATE=$(date +%Y%m%d)
docker exec kb-mysql mysqldump -u kb -pkb123456 knowledge_base | gzip > /mnt/e-data/minio/kb-system/backups/mysql/kb_${DATE}.sql.gz
# 清理30天前的备份
find /mnt/e-data/minio/kb-system/backups/mysql/ -name "*.sql.gz" -mtime +30 -delete
```

```bash
#!/bin/bash
# mongo-backup.sh
DATE=$(date +%Y%m%d)
docker exec kb-mongo mongodump --db knowledge_base --gzip --archive > /mnt/e-data/minio/kb-system/backups/mongo/kb_${DATE}.gz
find /mnt/e-data/minio/kb-system/backups/mongo/ -name "*.gz" -mtime +30 -delete
```

```bash
#!/bin/bash
# meili-backup.sh
DATE=$(date +%Y%m%d)

find /data/meilisearch/dumps/ -name "*.dump" -mtime +30 -delete
```

